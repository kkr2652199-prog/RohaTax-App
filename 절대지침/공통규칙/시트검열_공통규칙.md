```python
def _inspect_all_sheets(self, workbook):
    """공통 시트 검열 로직: 모든 업종에서 동일하게 적용"""
    sheet_results = []
    best_result = None
    best_score = -1
    
    # 1. 모든 시트 순회하며 평가
    for sheet_name in workbook.sheetnames:
        sheet = workbook[sheet_name]
        
        # 2. 각 시트 평가 (업종별 특화 로직 적용)
        sheet_result = self._evaluate_sheet(sheet, sheet_name, required_keywords)
        sheet_results.append(sheet_result)
    
    # 🚀 1순위 시트 Fast-Path: 1순위 시트 발견 시 즉시 반환
    priority_sheet = self._find_priority_sheet(workbook, required_keywords)
    if priority_sheet:
        self.logger.info(f"🎯 1순위 시트 발견! '{priority_sheet['sheet_name']}' - 즉시 선택")
        self.logger.info(f"🛑 2순위 프로세스 완전 중단 - 1순위 시트가 존재하므로 추가 검열 불필요")
        return priority_sheet  # Fast-Path: 1순위 시트 발견 시 즉시 반환
    
    # 3. 1순위 시트가 없을 경우 최고 점수 시트 선택
    best_result = max(sheet_results, key=lambda x: x['score'])
    return best_result
```

### **🔍 1순위 시트 판별 기준 (공통)**

1. **5형제 가족 존재**: 사업자번호, 대표자명, 주소, 이메일, 가맹점명 모두 존재
2. **데이터 품질**: 유효한 숫자 데이터 존재
3. **업종별 특화 조건**: 각 업종의 특화된 조건 추가 적용

### **📋 제목/부제목 행 건너뛰기 로직**

```python
def _detect_header_row(self, sheet) -> int:
    """공통 헤더 감지 로직: 제목/부제목 행을 건너뛰고 실제 헤더 찾기"""
    
    # 1. 제목/부제목 행 감지 및 제외
    title_rows = self._detect_title_rows(sheet)
    
    # 2. 실제 헤더 행 찾기 (제목 행 제외)
    for row_idx in range(len(sheet)):
        if row_idx in title_rows:
            continue  # 제목/부제목 행 건너뛰기
            
        # 3. 헤더 후보 평가
        header_score = self._evaluate_header_candidate(sheet, row_idx)
        if header_score > best_score:
            best_score = header_score
            best_header_row = row_idx
    
    return best_header_row
```

### **🔍 제목/부제목 패턴 감지**

```python
def _detect_title_rows(self, sheet) -> List[int]:
    """제목/부제목 행 감지"""
    title_rows = []
    
    for row_idx in range(min(10, len(sheet))):  # 상위 10행만 검사
        row_text = self._get_row_text(sheet, row_idx)
        
        # 제목/부제목 패턴 확인
        if self._is_title_pattern(row_text):
            title_rows.append(row_idx)
    
    return title_rows

def _is_title_pattern(self, text: str) -> bool:
    """제목/부제목 패턴 판별"""
    title_keywords = [
        '세금계산서', '발행용', '정산서', '내역서', '명세서',
        '년도', '월', '일', '기간', '요약', '합계'
    ]
    
    return any(keyword in text for keyword in title_keywords)
```

### **📋 사업자번호 기반 가족 통합**

```python
def _merge_families_by_business_number(self, families: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
    """공통 가족 통합 로직: 사업자번호 기반으로 분산된 가족 통합"""
    
    # 1. 사업자번호별로 그룹화
    business_groups = {}
    for family in families:
        biz_num = family.get('사업자등록번호', '')
        if biz_num:
            if biz_num not in business_groups:
                business_groups[biz_num] = []
            business_groups[biz_num].append(family)
    
    # 2. 각 그룹별로 통합
    merged_families = []
    for biz_num, group in business_groups.items():
        if len(group) > 1:  # 분산된 가족이 있는 경우
            merged_family = self._merge_family_group(group)
            merged_families.append(merged_family)
        else:
            merged_families.extend(group)
    
    return merged_families
```

### **🔍 통합 규칙**

1. **사업자번호 동일**: 같은 사업자번호를 가진 모든 행을 하나로 통합
2. **엄마 값 합산**: 모든 엄마(부가세) 값을 합산
3. **아빠 값 보정**: 엄마 합산 × 10 = 아빠 값 (세법 10% 규칙)

## 🎯 금지 컬럼 처리 공통 규칙

### **📋 조건부 금지 컬럼 로직**

```python
def _apply_forbidden_columns_logic(self, sheet_data, is_priority_sheet: bool):
    """공통 금지 컬럼 처리 로직"""
    
    forbidden_columns = ['콜수수료 공급가', '콜수수료 부가세']
    
    if is_priority_sheet:
        # 1순위 시트: 금지 컬럼 무시
        self.logger.info("🎯 1순위 시트: 금지 컬럼 무시 처리")
        return sheet_data
    else:
        # 2순위 시트: 금지 컬럼 제거
        self.logger.info("🚫 2순위 시트: 금지 컬럼 제거 처리")
        for col in forbidden_columns:
            if col in sheet_data.columns:
                sheet_data = sheet_data.drop(columns=[col])
        return sheet_data
```

### **🔍 적용 조건**

1. **1순위 시트**: 금지 컬럼 무시하고 데이터 추출 진행
2. **2순위 시트**: 금지 컬럼을 완전히 제거하고 데이터 추출 진행
3. **이산가족 통합**: 금지 컬럼이 있어도 사업자번호 기반 통합 수행
