```python
def load_guidelines(industry_type):
    """업종별 특화 지침 로딩"""
    guidelines = {
        'delivery': {
            'store_keywords': ['상호', '가게명', '매장명'],
            'amount_keywords': ['요금', '금액'],
            'vat_rate': 0.1
        },
        'general': {
            'store_keywords': ['업체명', '회사명'],
            'amount_keywords': ['금액', '비용'],
            'vat_rate': 0.1
        }
    }
    return guidelines.get(industry_type, {})
```

### **6단계: 데이터 추출**
```python
def extract_data_with_guidelines(file_data, guidelines):
    """지침에 따라 데이터 추출"""
    extracted_data = []
    
    for row in file_data:
        # 지침의 키워드로 추출
        store_name = extract_by_keywords(row, guidelines['store_keywords'])
        amount = extract_by_keywords(row, guidelines['amount_keywords'])
        vat = amount * guidelines['vat_rate']
        
        if store_name and amount:
            extracted_data.append({
                'store_name': store_name,
                'amount': amount,
                'vat': vat
            })
    
    return extracted_data
```

### **7단계: 템플릿 기입**
```python
def fill_template_with_guidelines(worksheet, extracted_data, supplier_info):
    """지침에 따라 템플릿에 기입"""
    start_row = 7  # 7행부터 시작
    
    # 공급자 정보 기입 (모든 행에 동일)
    for i, data in enumerate(extracted_data):
        row = start_row + i
        
        # 공급자 정보 (공급자정보_통합관리_절대지침.md 적용)
        worksheet[f'A{row}'] = '01'  # 전자세금계산서 종류
        worksheet[f'B{row}'] = '251001'  # 작성일자
        worksheet[f'C{row}'] = supplier_info['business_number']
        worksheet[f'E{row}'] = supplier_info['company_name']
        worksheet[f'F{row}'] = supplier_info['representative_name']
        worksheet[f'G{row}'] = supplier_info['address']
        worksheet[f'H{row}'] = supplier_info['business_type']
        worksheet[f'I{row}'] = supplier_info['business_category']
        worksheet[f'J{row}'] = supplier_info['email']
        
        # 공급받는자 정보 (공급받는자_통합관리_절대지침.md 적용)
        worksheet[f'K{row}'] = data['business_number']
        worksheet[f'M{row}'] = data['store_name']
        worksheet[f'N{row}'] = data['representative']
        worksheet[f'O{row}'] = data['address']
        worksheet[f'R{row}'] = data['email']
        worksheet[f'T{row}'] = data['amount']
        worksheet[f'U{row}'] = data['vat']
        
        # 절대값 기입
        worksheet[f'W{row}'] = '30'
        worksheet[f'BG{row}'] = '01'
```
