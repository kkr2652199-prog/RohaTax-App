### **📋 제목/부제목 행 건너뛰기 시스템**

```python
def _detect_header_row(self, sheet) -> int:
    """공통 헤더 감지 로직: 제목/부제목 행을 건너뛰고 실제 헤더 찾기"""
    
    best_score = -1
    best_header_row = 0
    
    # 1. 제목/부제목 행 감지 및 제외
    title_rows = self._detect_title_rows(sheet)
    self.logger.info(f"🔍 감지된 제목/부제목 행: {title_rows}")
    
    # 2. 실제 헤더 행 찾기 (제목 행 제외)
    for row_idx in range(min(1000, len(sheet))):  # 상위 1000행까지 스캔
        if row_idx in title_rows:
            continue  # 제목/부제목 행 건너뛰기
            
        # 3. 헤더 후보 평가
        header_score = self._evaluate_header_candidate(sheet, row_idx)
        if header_score > best_score:
            best_score = header_score
            best_header_row = row_idx
    
    self.logger.info(f"✅ 최적 헤더 행: {best_header_row}행 (점수: {best_score})")
    return best_header_row
```

### **🔍 제목/부제목 패턴 감지**

```python
def _detect_title_rows(self, sheet) -> List[int]:
    """제목/부제목 행 감지"""
    title_rows = []
    
    for row_idx in range(min(10, len(sheet))):  # 상위 10행만 검사
        row_text = self._get_row_text(sheet, row_idx)
        
        # 제목/부제목 패턴 확인
        if self._is_title_pattern(row_text):
            title_rows.append(row_idx)
            self.logger.info(f"📝 제목 행 감지: {row_idx}행 - '{row_text[:50]}...'")
    
    return title_rows

def _get_row_text(self, sheet, row_idx: int) -> str:
    """행의 텍스트 추출"""
    row_text = ""
    for col_idx in range(min(10, sheet.max_column)):  # 상위 10열만 검사
        cell_value = sheet.cell(row=row_idx+1, column=col_idx+1).value
        if cell_value:
            row_text += str(cell_value) + " "
    return row_text.strip()

def _is_title_pattern(self, text: str) -> bool:
    """제목/부제목 패턴 판별"""
    title_keywords = [
        '세금계산서', '발행용', '정산서', '내역서', '명세서',
        '년도', '월', '일', '기간', '요약', '합계', '총계',
        '가맹점', '세금계산서', '발행', '용', '서'
    ]
    
    # 제목 패턴 확인
    title_count = sum(1 for keyword in title_keywords if keyword in text)
    
    # 제목으로 판단하는 조건
    if title_count >= 2:  # 2개 이상의 제목 키워드
        return True
    
    if len(text) < 20 and title_count >= 1:  # 짧은 텍스트에 제목 키워드
        return True
    
    return False
```

### **🎯 헤더 후보 평가 시스템**

```python
def _evaluate_header_candidate(self, sheet, row_idx: int) -> float:
    """헤더 후보 평가"""
    
    score = 0.0
    required_keywords = ['가맹점명', '대표자명', '사업자등록번호', '사업장주소', '이메일']
    
    # 1. 필수 키워드 매칭 점수 (40%)
    matched_keywords = 0
    for col_idx in range(min(20, sheet.max_column)):  # 상위 20열 검사
        cell_value = str(sheet.cell(row=row_idx+1, column=col_idx+1).value or "")
        normalized_value = self.normalize_column_name(cell_value)
        
        for keyword in required_keywords:
            if keyword in normalized_value:
                matched_keywords += 1
                break
    
    keyword_score = (matched_keywords / len(required_keywords)) * 40
    score += keyword_score
    
    # 2. 데이터 밀도 점수 (30%)
    data_cells = 0
    total_cells = min(20, sheet.max_column)
    
    for col_idx in range(total_cells):
        cell_value = sheet.cell(row=row_idx+1, column=col_idx+1).value
        if cell_value and str(cell_value).strip():
            data_cells += 1
    
    density_score = (data_cells / total_cells) * 30
    score += density_score
    
    # 3. 데이터 품질 점수 (30%)
    quality_score = self._evaluate_data_quality(sheet, row_idx)
    score += quality_score
    
    return score

def _evaluate_data_quality(self, sheet, row_idx: int) -> float:
    """데이터 품질 평가"""
    
    quality_score = 0.0
    
    # 다음 행들의 데이터 품질 확인
    for next_row in range(row_idx + 1, min(row_idx + 5, len(sheet))):
        row_data = []
        for col_idx in range(min(10, sheet.max_column)):
            cell_value = sheet.cell(row=next_row+1, column=col_idx+1).value
            if cell_value:
                row_data.append(str(cell_value))
        
        # 데이터가 있는 행의 비율
        if len(row_data) > 0:
            quality_score += 6.0  # 각 행당 6점
    
    return min(quality_score, 30.0)  # 최대 30점
```

```python
def _handle_header_detection_errors(self, sheet, error_type: str):
    """헤더 감지 오류 처리"""
    
    if error_type == "no_header_found":
        self.logger.warning("⚠️ 헤더를 찾을 수 없음 - 기본값 사용")
        return 0  # 첫 번째 행을 헤더로 사용
    
    elif error_type == "multiple_headers":
        self.logger.warning("⚠️ 여러 헤더 후보 발견 - 최고 점수 선택")
        return self._select_best_header(sheet)
    
    elif error_type == "empty_sheet":
        self.logger.error("❌ 빈 시트 - 처리 중단")
        raise ValueError("빈 시트는 처리할 수 없습니다")
    
    else:
        self.logger.error(f"❌ 알 수 없는 오류: {error_type}")
        return 0
```

### **🔍 헤더 검증 시스템**

```python
def _validate_header(self, sheet, header_row: int) -> bool:
    """헤더 유효성 검증"""
    
    # 1. 헤더 행에 데이터가 있는지 확인
    has_data = False
    for col_idx in range(min(10, sheet.max_column)):
        cell_value = sheet.cell(row=header_row+1, column=col_idx+1).value
        if cell_value and str(cell_value).strip():
            has_data = True
            break
    
    if not has_data:
        self.logger.warning(f"⚠️ {header_row}행에 헤더 데이터 없음")
        return False
    
    # 2. 다음 행에 실제 데이터가 있는지 확인
    has_next_data = False
    for col_idx in range(min(5, sheet.max_column)):
        cell_value = sheet.cell(row=header_row+2, column=col_idx+1).value
        if cell_value and str(cell_value).strip():
            has_next_data = True
            break
    
    if not has_next_data:
        self.logger.warning(f"⚠️ {header_row+1}행에 데이터 없음")
        return False
    
    return True
```
