```python
def _inspect_all_sheets(self, workbook):
    """지능앱 핵심 기술: 모든 시트를 검열하여 최적의 시트 선택 (총배달 금액 우선 선점)"""
    sheet_results = []
    best_result = None
    best_score = -1
    
    # 1. 모든 시트 순회하며 총배달 금액 분석
    for sheet_name in workbook.sheetnames:
        sheet = workbook[sheet_name]
        
        # 2. 총배달 금액 컬럼 찾기 및 최대값 계산
        total_delivery_amount = self._find_max_delivery_amount(sheet)
        
        # 3. 각 시트 평가 (총배달 금액 가중치 적용)
        sheet_result = self._evaluate_sheet(sheet, sheet_name, required_keywords)
        
        # 4. 총배달 금액 보너스 점수 추가
        if total_delivery_amount > 0:
            delivery_bonus = min(total_delivery_amount / 1000000, 50)  # 최대 50점 보너스
            sheet_result['delivery_amount'] = total_delivery_amount
            sheet_result['delivery_bonus'] = delivery_bonus
            sheet_result['score'] += delivery_bonus
        
        sheet_results.append(sheet_result)
    
    # 🚀 1순위 시트 Fast-Path: 1순위 시트 발견 시 즉시 반환
    priority_sheet = self._find_priority_sheet(workbook, required_keywords)
    if priority_sheet:
        self.logger.info(f"🎯 1순위 시트 발견! '{priority_sheet['sheet_name']}' (아빠 금액: {priority_sheet['max_dad_amount']:,.0f}원) - 즉시 선택")
        self.logger.info(f"🛑 2순위 프로세스 완전 중단 - 1순위 시트가 존재하므로 추가 검열 불필요")
        return priority_sheet  # Fast-Path: 1순위 시트 발견 시 즉시 반환
    
    # 5. 총배달 금액이 가장 높은 시트 우선 선점
    best_result = max(sheet_results, key=lambda x: x.get('delivery_amount', 0))
    
    return best_result

def _find_max_delivery_amount(self, sheet):
    """총배달 금액 컬럼에서 최대값 찾기"""
    max_amount = 0
    
    # 총배달 금액 관련 키워드
    delivery_keywords = ['총배달', '배달금액', '총금액', '합계', '총합', '배달요금', '총배달요금']
    
    for row in sheet.iter_rows():
        for cell in row:
            cell_value = str(cell.value).lower().strip()
            
            # 헤더에서 총배달 금액 컬럼 찾기
            if any(keyword in cell_value for keyword in delivery_keywords):
                col_idx = cell.column
                
                # 해당 컬럼의 모든 숫자 값 중 최대값 찾기
                for data_row in sheet.iter_rows(min_row=cell.row+1, min_col=col_idx, max_col=col_idx):
                    for data_cell in data_row:
                        if isinstance(data_cell.value, (int, float)) and data_cell.value > max_amount:
                            max_amount = data_cell.value
                
                break
    
    return max_amount
```

```python
def _count_matched_fields(self, sheet, row, max_col, required_keywords):
    """지능앱 기술: 5개 핵심 필드 정밀 매칭으로 헤더 검증"""
    matched_fields = set()  # 중복 매칭 방지
    
    for col in range(1, max_col + 1):
        cell_text = str(cell_value).lower().strip()
        
        # 각 필드 타입별 정밀 키워드 매칭
        for field_type, keywords in required_keywords.items():
            if field_type in matched_fields:
                continue  # 이미 매칭된 필드는 건너뛰기
                
            for keyword in keywords:
                if keyword.lower() in cell_text:
                    matched_fields.add(field_type)
                    break
```

#### **🔍 헤더 감지 개선 사항**
- **스캔 범위 확장**: 상단 30행까지 스캔 (기존 10행)
- **정밀 매칭**: 5개 핵심 필드별 키워드 매칭 강화
- **중복 방지**: set()을 사용하여 중복 매칭 방지
- **다중 기준 선택**: 매칭 필드 수 → 헤더 점수 → 행 번호 순으로 선택
- **추가 검증**: 최소 5개 이상의 비어있지 않은 셀 확인

#### **📈 헤더 감지 결과 활용**
- **Excel 파일**: 30행까지 스캔하여 최적 헤더 선택
- **CSV 파일**: 10행까지 스캔하여 최적 헤더 선택
- **로깅 강화**: 헤더 후보별 상세 정보 기록

- 스킵 건수는 로그로만 기록(현 단계 UI 표시 없음)
- 참고 규칙(후속 예정): 합계 열 기준 좌/우 1열 범위 내 부가세 헤더 인접 탐색


```python
                break

            cell_value = sheet_data.iloc[row_idx, candidate_col]
            if isinstance(cell_value, (int, float)) and cell_value > 0:
                numeric_count += 1
        
        if numeric_count > max_numeric_data:
            max_numeric_data = numeric_count
            best_dad_column = candidate_col
    
    if best_dad_column is None:
        return None, "아빠(총배달금액) 컬럼을 찾을 수 없습니다"
```

```python
                    found_columns += 1
                    break
        
        if found_columns >= 5:
            return {
                'sheet_name': sheet_name,
                'header_row': header_row,
                'found_columns': found_columns
            }
    
    return None
```

```python
def detect_header_row(sheet):
    """헤더 행 자동 감지 (데이터 밀도 기반)"""
    max_density = 0
    best_row = 1
    
    for row in range(1, min(10, sheet.max_row + 1)):  # 처음 10행만 검사
        density = calculate_data_density(sheet, row)
        if density > max_density:
            max_density = density
            best_row = row
    
    return best_row

def calculate_data_density(sheet, row):
    """행의 데이터 밀도 계산"""
    text_count = 0
    total_cells = 0
    
    for col in range(1, sheet.max_column + 1):
        cell_value = sheet.cell(row=row, column=col).value
        if cell_value:
            total_cells += 1
            if isinstance(cell_value, str):
                text_count += 1
    
    return text_count / total_cells if total_cells > 0 else 0
```

## 🎯 2순위 시트 처리 규칙 (배달대행사 특화)

**1순위 시트가 존재하지 않는 파일은 2순위가 1순위 검열방식을 통해 파일을 추출하며, 이산가족상봉 프로세스는 중단하고 사업자등록번호로만 통합합니다.**

### **🔍 배달대행사 2순위 시트 감지 조건**

1. **사업자등록번호 빈 비율**: 전체 행 중 50% 이상이 사업자등록번호가 비어있음
2. **분산된 가족 패턴**: 같은 사업자번호를 가진 여러 행이 있지만 상호나 대표자명이 다름
3. **배달대행사 특화 컬럼**: 총배달금액, 배달수수료 등 배달대행사 특화 컬럼이 있는 경우
4. **헤더 매핑 오류**: 사업자등록번호가 이메일로 잘못 인식되는 패턴

### **🔄 배달대행사 2순위 시트 처리 원칙**

1. **2순위 시트 = 1순위 검열방식 적용**
   - 1순위 시트가 없으면 2순위 시트를 1순위 방식으로 처리
   - 동일한 헤더 매핑 및 추출 로직 사용

2. **이산가족상봉 프로세스 중단**
   - 분산된 가족을 찾아서 통합하는 복잡한 프로세스 중단
   - 대표자명 중복 여부 무시
   - 상호명 차이 무시

3. **사업자등록번호로만 통합**
   - 사업자등록번호가 동일한 행들만 통합
   - 사업자등록번호가 다르면 별도 행으로 처리
   - 금액 합산: 공급가액, 부가세, 요금합계 모두 합산

   - **특징**: 1사람이 여러 개 보유 가능

   - **검증 규칙**: `@` 문자가 포함된 형식만 이메일로 인식
   - **도메인**: `naver.com`, `gmail.com`, `daum.net` 등 유효한 도메인

### **🎯 배달대행사 사업자등록번호 기반 통합 처리**

2순위 시트에서 감지된 분산된 가족들을 배달대행사 특화 규칙으로 사업자등록번호로만 통합합니다:

1. **사업자번호 우선 통합**: 같은 사업자번호를 가진 행들을 우선적으로 통합
2. **배달대행사 금액 합산**: 총배달금액, 배달수수료, 공급가액, 부가세를 모두 합산
3. **정보 통합**: 상호명과 대표자명은 가장 긴 것을 사용
4. **개별 처리**: 사업자번호가 없는 행은 그대로 유지

```python
def _integrate_scattered_families_delivery(self, recipients: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
    """배달대행사 사업자등록번호 기반 통합 처리 (2순위 시트 전용)"""
    # 사업자번호별로 그룹화
    business_groups = {}
    for recipient in recipients:
        business_num = recipient.get('사업자등록번호', '')
        if business_num and business_num != '':
            if business_num not in business_groups:
                business_groups[business_num] = []
            business_groups[business_num].append(recipient)
    
    # 통합된 결과
    integrated_recipients = []
    
    # 사업자번호별로 통합 (대표자명 중복 무시)
    for business_num, group in business_groups.items():
        if len(group) > 1:
            # 배달대행사 특화 통합된 가족 생성 (사업자번호 기준)
            integrated_family = self._create_integrated_family_by_business_number_delivery(group)
            integrated_recipients.append(integrated_family)
        else:
            # 단일 건은 그대로 추가
            integrated_recipients.extend(group)
    
    return integrated_recipients
```
